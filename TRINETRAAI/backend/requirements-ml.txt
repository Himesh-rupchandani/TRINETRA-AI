# TRINETRA AI backend — computer-vision extras
# =============================================
# Installs the core API set (via -r) plus everything the frame-level pipeline
# needs: OpenCV decoding, YOLO11 vehicle + number-plate detection, and OCR.
# Required for live camera ingestion, /api/cameras/{id}/live, on-demand
# detection, video uploads and ANPR. Not required to serve the data API.
-r requirements.txt

# Real-time vehicle detection (YOLO11) for the live view. CPU-only torch is
# enough: pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu
ultralytics>=8.3.0
# ANPR/OCR for uploaded CCTV videos (bundled ONNX models — works fully offline).
# The pipeline degrades gracefully to "Unknown" plates when this is absent.
rapidocr-onnxruntime>=1.0.0
numpy>=1.26.0
pillow>=10.2.0
# Server-safe cv2 provider. OpenCV is repaired after dependency installation by:
#   python ../../scripts/ensure_headless_opencv.py
# (ultralytics declares GUI opencv-python, so listing the headless wheel here
# alone is not enough to make pip choose the server-safe build.)
opencv-python-headless>=4.9.0
# Migrations helper for long-lived deployments (schema is otherwise created and
# auto-migrated at boot by app/database/database.py).
alembic>=1.13.1
